  sudo mount -t vboxsf snauto /mnt/snauto
  sudo cp -a /mnt/snauto /root/sn-lsp-auto
  cd /root/sn-lsp-auto
  sudo chmod +x install_sn_lsp.sh configure_sn_lsp.sh
  sudo ./install_sn_lsp.sh --pkg-dir ./packages-astra --license ./packages-astra/26962ED_key.lic
